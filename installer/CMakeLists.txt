cmake_minimum_required(VERSION 3.21)
project(dan_player_installer LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The installer uses documented Windows APIs.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(dan_installer_core STATIC native/installer_core.cpp)
target_include_directories(dan_installer_core PUBLIC native)
target_compile_definitions(dan_installer_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(dan_installer_core PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_core PUBLIC bcrypt)
add_library(dan_installer_native SHARED native/installer_bridge.cpp native/branding.cpp native/material_controls.cpp native/update_gate.cpp)
target_compile_options(dan_installer_native PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_native PRIVATE dan_installer_core shell32 ole32 advapi32 gdiplus gdi32 comctl32 dwmapi uxtheme)
add_library(dan_installer_native_qa SHARED native/installer_bridge.cpp native/branding.cpp native/material_controls.cpp native/update_gate.cpp tests/inno_ui_render.cpp)
target_compile_definitions(dan_installer_native_qa PRIVATE DAN_INSTALLER_TEST_BUILD)
target_compile_options(dan_installer_native_qa PRIVATE /W4 /WX /wd4505 /utf-8 /EHsc)
target_link_libraries(dan_installer_native_qa PRIVATE dan_installer_core shell32 ole32 advapi32 gdiplus gdi32 comctl32 dwmapi uxtheme)
add_executable(dan_installer_cleanup WIN32 native/cleanup.cpp)
target_compile_options(dan_installer_cleanup PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_cleanup PRIVATE dan_installer_core shell32 version)
add_executable(dan_installer_core_test tests/installer_core_test.cpp)
target_compile_options(dan_installer_core_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_core_test PRIVATE dan_installer_core)
enable_testing()
add_test(NAME dan_installer_core_test COMMAND dan_installer_core_test
  "${CMAKE_BINARY_DIR}/test-sandbox")
add_executable(dan_installer_controls_test tests/installer_controls_test.cpp)
target_compile_options(dan_installer_controls_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(dan_installer_controls_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dan_installer_controls_test PRIVATE dan_installer_native gdi32)
add_test(NAME dan_installer_controls_test COMMAND dan_installer_controls_test
  "${CMAKE_CURRENT_SOURCE_DIR}/../assets/fonts/PingFangSC-Regular.ttf"
  "${CMAKE_BINARY_DIR}/test-sandbox")
add_executable(dan_installer_cleanup_fixture WIN32 tests/cleanup_fixture.cpp tests/cleanup_fixture.rc)
target_compile_options(dan_installer_cleanup_fixture PRIVATE /W4 /WX /utf-8 /EHsc)
add_executable(dan_installer_cleanup_test tests/installer_cleanup_test.cpp)
target_compile_options(dan_installer_cleanup_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_cleanup_test PRIVATE dan_installer_core)
add_test(NAME dan_installer_cleanup_test COMMAND dan_installer_cleanup_test
  "${CMAKE_BINARY_DIR}/test-sandbox" "$<TARGET_FILE:dan_installer_cleanup>"
  "$<TARGET_FILE:dan_installer_cleanup_fixture>")
add_executable(dan_installer_private_desktop_runner tests/inno_private_desktop_runner.cpp)
target_compile_options(dan_installer_private_desktop_runner PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(dan_installer_private_desktop_runner PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(dan_installer_brand_test tests/installer_brand_test.cpp)
target_compile_options(dan_installer_brand_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(dan_installer_brand_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dan_installer_brand_test PRIVATE dan_installer_native gdi32)
add_test(NAME dan_installer_brand_test COMMAND dan_installer_brand_test
  "${CMAKE_CURRENT_SOURCE_DIR}/../assets/images/RCE_logo_transparent.png"
  "${CMAKE_CURRENT_SOURCE_DIR}/../assets/branding/danruguo_light.png"
  "${CMAKE_CURRENT_SOURCE_DIR}/../assets/images/RCE_logo_white.png"
  "${CMAKE_CURRENT_SOURCE_DIR}/../assets/branding/danruguo_dark.png")
add_executable(dan_installer_update_gate_test tests/installer_update_gate_test.cpp native/update_gate.cpp)
target_compile_options(dan_installer_update_gate_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_update_gate_test PRIVATE dan_installer_core)
add_test(NAME dan_installer_update_gate_test COMMAND dan_installer_update_gate_test
  "${CMAKE_BINARY_DIR}/test-sandbox")
add_executable(dan_installer_inno_update_test tests/inno_update_test.cpp)
target_compile_options(dan_installer_inno_update_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_inno_update_test PRIVATE dan_installer_core)
