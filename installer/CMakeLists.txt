cmake_minimum_required(VERSION 3.21)
project(dan_player_installer LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The installer uses documented Windows APIs.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(dan_installer_core STATIC native/installer_core.cpp)
target_include_directories(dan_installer_core PUBLIC native)
target_compile_definitions(dan_installer_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(dan_installer_core PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_core PUBLIC bcrypt)
add_library(dan_installer_native SHARED native/installer_bridge.cpp native/branding.cpp native/progress.cpp native/material_controls.cpp native/update_gate.cpp)
target_compile_options(dan_installer_native PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_native PRIVATE dan_installer_core shell32 ole32 advapi32 gdiplus gdi32 comctl32 dwmapi uxtheme)
add_executable(dan_installer_cleanup WIN32 native/cleanup.cpp)
target_compile_options(dan_installer_cleanup PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(dan_installer_cleanup PRIVATE dan_installer_core shell32 version)
